### 2) Determination (기본값 설정)
- 사용자가 생성(Create) 화면에서 데이터를 입력하고 저장(Save)하는 순간, 시스템이 자동으로 기본값을 넣어
```abap
" Behavior Definition Root
determination <메서드명> on save {
    <create / update>;    " 실행 시점
    field <필드>;          " 해당 필드 변경 시 실행
}

" Class
METHODS <메서드명>        " Behavior Handler
  FOR DETERMINE ON SAVE
  IMPORTING keys FOR <Entity>~<메서드명>.

METHOD <메서드명>.        " Implementation
  MODIFY ENTITIES OF <Behavior Root> IN LOCAL MODE
    ENTITY <Entity>
    UPDATE FIELDS ( <필드> )
    WITH VALUE #(
      (
        %tky = <키>
        <필드> = <기본값>
      )
    ).
ENDMETHOD.
```

### 3) Validation (데이터 검증)
- 저장(Create/Update) 시 데이터의 유효성을 검사하고, 조건에 맞지 않으면 저장을 막음
- 저장 시 `Determination → Validation → DB` 저장 순으로 실행
```abap
" Behavior Definition Root
validation <메서드명> on save { <create / update>; }

" Class
METHODS <메서드명>        " Behavior Handler
  FOR VALIDATE ON SAVE
  IMPORTING keys FOR <Entity>~<메서드명>.

" Implementation
METHOD <메서드명>.
  READ ENTITIES OF <Behavior Root>
    ENTITY <Entity>
    ALL FIELDS
    WITH CORRESPONDING #( keys )
    RESULT DATA(<조회결과>).

  " 데이터 검증
  IF <조건>.
    APPEND VALUE #(
      %tky = <조회결과>-%tky
      %msg = new_message(
                id       = '<메시지클래스>'
                number   = '<번호>'
                severity = if_abap_behv_message=>severity-error )
    ) TO reported-<Entity>.

    APPEND VALUE #(
      %tky = <조회결과>-%tky
    ) TO failed-<Entity>.
  ENDIF.
ENDMETHOD.
```

### 4) Instance Authorization (데이터별 권한 제어)
```abap
" Behavior Definition Root
update ( features : instance );              " Update 기능 제어
field ( features : instance ) <필드>;        " 특정 필드 제어

" Class
METHODS get_instance_features                " Behavior Handler
  FOR INSTANCE FEATURES
  IMPORTING keys
            REQUEST requested_features
  FOR <Entity>
  RESULT result.

" Implementation
METHOD get_instance_features.
  READ ENTITIES OF <Behavior Root>
    ENTITY <Entity>
    FIELDS ( <조건 필드> )
    WITH CORRESPONDING #( keys )
    RESULT DATA(<조회결과>).

  DATA <결과> LIKE LINE OF result.

  LOOP AT <조회결과> INTO DATA(<데이터>).
    <결과>-%tky = <데이터>-%tky.

    CASE <데이터>-<조건 필드>.
      WHEN <조건1>.
        <결과>-%features-%update = if_abap_behv=>fc-o-disabled.     " 행 전체 수정 불가
      WHEN <조건2>.
        <결과>-%field-<필드> = if_abap_behv=>fc-f-read_only.         " 특정 필드만 수정 불가
    ENDCASE.

    APPEND <결과> TO result.
  ENDLOOP.
ENDMETHOD.
```

```abap
        ls_result-%tky = ls_emp-%tky.
        " Waers : readonly

          FOR key IN keys (
              %tky = key-%tky
```

```abap
managed implementation in class zbp_r_std_a06d3 unique;
strict ( 2 );

define behavior for ZR_STD_A06D3 alias Student
persistent table zta06stdd3
lock master
authorization master ( instance )
etag master LastChangedAt
{
  create;
  update;
  delete;
  field ( numbering : managed ) StdnoUuid;
  field ( mandatory : create ) Stdno;
  field ( readonly : update ) Stdno;
  field ( readonly ) StdStop, StdnoUuid;

  determination set_start_date on save { create; }
  validation check_pernr on save { create; update; }
  validation check_stdno on save { create; }

  action set_stop_stdno result [0..1] $self;

  mapping for zta06stdd3 corresponding
    {
      StdnoUuid     = stdno_uuid;
      StartDate     = start_date;
      StdStop       = std_stop;
      CreationAt    = creation_at;
      CreatedBy     = created_by;
      LastChangedAt = last_changed_at;
      LastChangedBy = last_changed_by;
    }
}
```
- set_start_date, check_pernr, check_stdno (클래스)
```abap
CLASS lhc_Student DEFINITION INHERITING FROM cl_abap_behavior_handler.
  PRIVATE SECTION.
    METHODS get_instance_authorizations FOR INSTANCE AUTHORIZATION
      IMPORTING keys REQUEST requested_authorizations FOR Student RESULT result.
    METHODS set_stop_stdno FOR MODIFY
      IMPORTING keys FOR ACTION student~set_stop_stdno RESULT result.
    METHODS set_start_date FOR DETERMINE ON SAVE
      IMPORTING keys FOR student~set_start_date.
    METHODS check_pernr FOR VALIDATE ON SAVE
      IMPORTING keys FOR student~check_pernr.
    METHODS check_stdno FOR VALIDATE ON SAVE
      IMPORTING keys FOR student~check_stdno.
ENDCLASS.

CLASS lhc_Student IMPLEMENTATION.
  METHOD get_instance_authorizations.
  ENDMETHOD.

  METHOD set_stop_stdno.
    READ ENTITIES OF zr_std_a06d3 IN LOCAL MODE
        ENTITY Student
        ALL FIELDS WITH CORRESPONDING #( keys )
        RESULT DATA(result_std).

    LOOP AT result_std INTO DATA(ls_std) WHERE StdStop = 'X'.
      APPEND VALUE #(
        %tky = ls_std-%tky
        %msg = new_message(
            id = 'ZMCA06_STD'
            number = '001'
            v1 = ls_std-Sname
            v2 = ls_std-Stdno
            severity = if_abap_behv_message=>severity-error
        )
      ) TO reported-student.

      APPEND VALUE #(
        %tky = ls_std-%tky
      ) TO failed-student.
    ENDLOOP.

    MODIFY ENTITIES OF zr_std_a06d3 IN LOCAL MODE
        ENTITY Student
        UPDATE FIELDS ( StdStop )
        WITH VALUE #(
            FOR std IN result_std
            WHERE ( StdStop IS INITIAL ) (
                %tky = std-%tky
                StdStop = 'X'
            )
        )
        FAILED DATA(failed_std)
        REPORTED DATA(reported_std).

    READ ENTITIES OF zr_std_a06d3
        ENTITY Student
        ALL FIELDS WITH CORRESPONDING #( keys )
        RESULT DATA(final_result_std).

    result = VALUE #(
        FOR std IN final_result_std
        (
            %tky = std-%tky
            %param = std
        )
    ).

  ENDMETHOD.

  METHOD set_start_date.
    DATA(ls_key) = VALUE #( keys[ 1 ] ).

    MODIFY ENTITIES OF zr_std_a06d3 IN LOCAL MODE
        ENTITY Student
        UPDATE FIELDS ( StartDate )
        WITH VALUE #( ( %tky = ls_key-%tky
                        StartDate = cl_abap_context_info=>get_system_date( ) ) )
        FAILED DATA(ls_failed).
  ENDMETHOD.

  METHOD check_pernr.
    READ ENTITIES OF zr_std_a06d3 IN LOCAL MODE
        ENTITY Student
        FIELDS ( Pernr )
        WITH CORRESPONDING #( keys MAPPING %tky = %key )
        RESULT DATA(result_std).

    DATA lv_flag TYPE abap_bool.
    lv_flag = abap_false.

    IF result_std[ 1 ]-Pernr IS NOT INITIAL.
      SELECT FROM zta06emp_d01
          FIELDS ( pernr )
          INTO TABLE @DATA(lt_emp).

      LOOP AT lt_emp INTO DATA(ls_emp).
        IF result_std[ 1 ]-Pernr = ls_emp-pernr.
          lv_flag = abap_true.
          EXIT.
        ENDIF.
      ENDLOOP.

      IF lv_flag = abap_false.
        APPEND VALUE #(
            %tky = result_std[ 1 ]-%tky
            %msg = new_message(
                id = 'ZMCA06'
                number = '003'
                v1 = result_std[ 1 ]-Pernr
                severity = if_abap_behv_message=>severity-error
            )
        ) TO reported-student.

        APPEND VALUE #(
            %tky = result_std[ 1 ]-%tky
        ) TO failed-student.
      ENDIF.
    ENDIF.
  ENDMETHOD.

  METHOD check_stdno.
    SELECT FROM zta06stdd3
        FIELDS ( stdno )
        INTO TABLE @DATA(lt_std).

    READ ENTITIES OF zr_std_a06d3 IN LOCAL MODE
        ENTITY Student
        ALL FIELDS WITH CORRESPONDING #( keys MAPPING %tky = %tky )
        RESULT DATA(result_stds).

    DATA lv_flag TYPE abap_bool.
    lv_flag = abap_false.

    LOOP AT lt_std INTO DATA(ls_std).
      IF result_stds[ 1 ]-Stdno = ls_std-stdno.
        lv_flag = abap_true.
        EXIT.
      ENDIF.
    ENDLOOP.

    IF lv_flag = abap_true.
      APPEND VALUE #(
          %tky = result_stds[ 1 ]-%tky
          %msg = new_message(
              id = 'ZMCA06'
              number = '004'
              v1 = result_stds[ 1 ]-Stdno
              severity = if_abap_behv_message=>severity-error
          )
      ) TO reported-student.

      APPEND VALUE #(
          %tky = result_stds[ 1 ]-%tky
      ) TO failed-student.
    ENDIF.
  ENDMETHOD.
ENDCLASS.
```
