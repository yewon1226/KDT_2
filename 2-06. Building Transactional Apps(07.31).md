### 2) 부모 Root 뷰 생성
- ZR_A06_SKILL
<img width="508" height="338" alt="image" src="https://github.com/user-attachments/assets/1fbdddc5-64a8-4b24-b167-40ccfd5f8b23" />

```abap
@AccessControl.authorizationCheck: #NOT_REQUIRED
@EndUserText.label: '[A06] 기술(부모) Root View'
@Metadata.ignorePropagatedAnnotations: true
define root view entity ZR_A06_SKILL as select from zta06skill_d03
composition [0..*] of ZI_A06_SKILL_TEXT as _ChildText
{
    key skill_uuid as SkillUuid,
    skill_code as SkillCode,
    @Semantics.amount.currencyCode: 'Waers'
    max_extra_price as MaxExtraPrice,
    @Semantics.amount.currencyCode: 'Waers'
    extra_price as ExtraPrice,
    waers as Waers,
    @Semantics.user.createdBy: true
    created_by as CreatedBy,
    @Semantics.systemDateTime.createdAt: true
    creation_at as CreationAt,
    @Semantics.user.lastChangedBy: true
    last_changed_by as LastChangedBy,
    @Semantics.systemDate.lastChangedAt: true
    last_changed_at as LastChangedAt,
    @Semantics.systemDate.localInstanceLastChangedAt: true
    local_last_changed_at as LocalLastChangedAt,
    _ChildText
}
```
